```bash
cd video-asia && npm install
node kiem-tra.mjs                               # soát kịch bản: phụ đề khớp lời, không cụm từ cấm
node tao-giong.mjs                              # lên giọng (cần ELEVENLABS_API_KEY; chỉ gọi lại cảnh đổi lời)
node kiem-giong.mjs                             # nghe lại bằng Speech-to-Text, so với lời gốc
node dung.mjs && node render.mjs                # dựng + xuất xuat/<ma>.mp4
```
